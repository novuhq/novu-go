## List

### Example Usage

<!-- UsageSnippet language="go" operationID="ActivityController_getLogs" method="get" path="/v1/activity/requests" -->

	"github.com/novuhq/novu-go/v3"
	"github.com/novuhq/novu-go/v3/models/operations"
	"log"
)

func main() {
    ctx := context.Background()

    s := v3.New(
        v3.WithSecurity("YOUR_SECRET_KEY_HERE"),
    )

    res, err := s.Activity.Requests.List(ctx, operations.ActivityControllerGetLogsRequest{
        StatusCodes: []float64{
            200.0,
            404.0,
            500.0,
        },
        CreatedGte: v3.Pointer[float64](1640995200.0),
    })
    if err != nil {
        log.Fatal(err)
    }
    if res.GetRequestsResponseDto != nil {
        // handle response
    }
}
